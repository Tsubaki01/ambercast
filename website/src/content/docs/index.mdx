import { Card, CardGrid } from '@astrojs/starlight/components';

Like an insect preserved in amber, your test's intent is cast once and kept intact — no matter how the surface changes.

<CardGrid stagger>
	<Card title="Zero AI calls on replay" icon="approve-check">
		Generate once, replay forever. `ambercast run` replays the committed plan deterministically — no AI call unless a step's cached grounding goes missing.
	</Card>
	<Card title="Lockfile-like plans" icon="document">
		`generate` writes a plan and a grounding cache as plain, git-diffable JSON. Commit and review them like a lockfile.
	</Card>
	<Card title="Self-healing" icon="rocket">
		When the UI drifts, `ambercast heal` re-resolves, repairs, or regenerates the affected plan steps — and asks for confirmation before writing.
	</Card>
	<Card title="Bring your own AI provider" icon="setting">
		No managed credentials. ambercast drives an already-authenticated Claude Code CLI or Codex CLI that you install yourself.
	</Card>
</CardGrid>
